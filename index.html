async function askGeminiIfConfigured(userQuestion, localContextSummary) { const key = (db.settings && db.settings.geminiApiKey) ? db.settings.geminiApiKey.trim() : ""; if (!key) return { ok: false, reason: "Hakuna Gemini API Key iliyohifadhiwa." }; const prompt = `Wewe ni mshauri mahiri wa biashara kwa Faustin Stationery Services (Tanzania). Jibu kwa Kiswahili kifupi, wazi na chenye vitendo.\n\nTakwimu za sasa:\n${localContextSummary}\n\nSwali la mtumiaji: ${userQuestion}`; const modelsToTry = ["gemini-2.5-flash", "gemini-2.5-flash-lite", "gemini-flash-latest"]; let lastErr = "Imeshindwa kupata jibu kutoka Gemini."; for (const modelName of modelsToTry) { for (let attempt = 1; attempt <= 2; attempt++) { try { const endpoint = `https://generativelanguage.googleapis.com/v1beta/models/${modelName}:generateContent?key=${encodeURIComponent(key)}`; const res = await fetch(endpoint, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ contents: [{ parts: [{ text: prompt }] }] }) }); const data = await res.json(); if (!res.ok) { const msg = data?.error?.message || `HTTP ${res.status}`; lastErr = `${msg} (${modelName})`; const isOverloaded = res.status === 429 || res.status === 503 || /high demand|overloaded|unavailable/i.test(msg); if (isOverloaded && attempt === 1) { await new Promise(r => setTimeout(r, 700)); continue; } break; } const text = data?.candidates?.[0]?.content?.parts?.[0]?.text; if (text) { const suffix = attempt > 1 ? `${modelName} • retry` : modelName; return { ok: true, text: text.trim(), model: suffix }; } } catch (err) { lastErr = `${err.message || "Tatizo la mtandao au kivinjari."} (${modelName})`; if (attempt === 1) { await new Promise(r => setTimeout(r, 700)); continue; } break; } } } return { ok: false, reason: lastErr }; }
